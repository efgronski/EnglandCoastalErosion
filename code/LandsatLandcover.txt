// Cloud/shadow mask for Landsat L2 using QA_PIXEL bits
// Bits: 3 = cloud shadow, 4 = snow, 5 = cloud, 7 = cirrus
function maskL2_QA(image) {
  var qa = image.select('QA_PIXEL');
  var cloudShadowBitMask = 1 << 3;
  var snowBitMask        = 1 << 4;
  var cloudBitMask       = 1 << 5;
  var cirrusBitMask      = 1 << 7;

  var mask = qa.bitwiseAnd(cloudShadowBitMask).eq(0)
    .and(qa.bitwiseAnd(snowBitMask).eq(0))
    .and(qa.bitwiseAnd(cloudBitMask).eq(0))
    .and(qa.bitwiseAnd(cirrusBitMask).eq(0));

  // Scale optical SR bands: scale = 0.0000275, offset = -0.2 (per USGS spec)
  var optical = image.select(['SR_B1','SR_B2','SR_B3','SR_B4','SR_B5','SR_B7'])
                    .multiply(0.0000275).add(-0.2);

  return image.addBands(optical, null, true).updateMask(mask);
}

// Remote Sensing Classification using ESA WorldCover (all 11 classes)
// Labels: ESA WorldCover v200 2021 codes remapped to 0..10
// ============================================================

function getPredictorBands(image) {
  var bandNames = image.bandNames();
  var hasB6 = bandNames.contains('SR_B6');

  // Landsat 8/9
  var l8Bands = [
    'SR_B2','SR_B3','SR_B4','SR_B5','SR_B6','SR_B7'
  ];

  // Landsat 7 (no SR_B6)
  var l7Bands = [
    'SR_B2','SR_B3','SR_B4','SR_B5','SR_B7'
  ];

  return ee.List(
    ee.Algorithms.If(hasB6, l8Bands, l7Bands)
  );
}

function scaleLandsatSR(img) {
  return img; // already scaled in maskL2_QA
}

function runLandcoverClassification(landsatComposite, year) {

  // 1) Study area
  var roi = yorkshireGeometry;

  // 2) Use PROVIDED composite (this is the key change)
  var image = scaleLandsatSR(landsatComposite);

  var ndvi = image.normalizedDifference(['SR_B5','SR_B4']).rename('NDVI');
  var ndwi = image.normalizedDifference(['SR_B3','SR_B5']).rename('NDWI');
  image = image.addBands([ndvi, ndwi]);

  var srBands = getPredictorBands(image);
  var predictors = image.select(srBands.add('NDVI').add('NDWI'));

  // 3) ESA WorldCover (unchanged)
  var wc = ee.Image('ESA/WorldCover/v200/2021');
  var wcBand = 'Map';

  var wcCodes = [10,20,30,40,50,60,70,80,90,95,100];
  var wcNames = [
    'Tree cover','Shrubland','Grassland','Cropland','Built-up',
    'Bare / sparse','Snow & ice','Water','Herbaceous wetland',
    'Mangroves','Moss & lichen'
  ];
  var wcPalette = [
    '006400','ffbb22','ffff4c','f096ff','fa0000',
    'b4b4b4','f0f0f0','0064c8','0096a0','00cf75','fae6a0'
  ];

  var wcCodesList = ee.List(wcCodes);
  var toIdx = ee.List.sequence(0, wcCodes.length - 1);
  var ones = ee.List.repeat(1, wcCodesList.length());

  var keptMask = wc.select(wcBand)
    .remap(wcCodesList, ones)
    .neq(0);

  var labels = wc.select(wcBand)
    .updateMask(keptMask)
    .remap(wcCodesList, toIdx)
    .rename('landcover')
    .toInt();

  // 4) Stratified sampling
  var sample = predictors.addBands(labels).stratifiedSample({
    numPoints: 2000,
    classBand: 'landcover',
    region: roi,
    scale: 10,
    geometries: true,
    seed: 42
  });

  // 5) Train/test split
  var withRand = sample.randomColumn('rand', 42);
  var train = withRand.filter(ee.Filter.lt('rand', 0.7));
  var test = withRand.filter(ee.Filter.gte('rand', 0.7));

  // 6) Train RF
  var clf = ee.Classifier.smileRandomForest({
    numberOfTrees: 300,
    seed: 42
  }).train({
    features: train,
    classProperty: 'landcover',
    inputProperties: predictors.bandNames()
  });

  // 7) Classify
  var classified = predictors.classify(clf)
    .rename('classification')
    .set('year', year);

  // 8) Accuracy
  var testPred = test.classify(clf);
  var cm = testPred.errorMatrix('landcover', 'classification');

  print('=== Year:', year, '===');
  print('Confusion Matrix:', cm);
  print('Overall Accuracy:', cm.accuracy());
  print('Kappa:', cm.kappa());

  // 9) Area by class
  var areaImage = ee.Image.pixelArea().addBands(classified);
  var areaStats = areaImage.reduceRegion({
    reducer: ee.Reducer.sum().group({
      groupField: 1,
      groupName: 'class'
    }),
    geometry: roi,
    scale: 30,
    maxPixels: 1e10
  });

  print('Area (m^2) by class:', areaStats);
  
  // Convert groups (List of Dictionaries) to a FeatureCollection for robust filtering/sorting
  var groups = ee.List(areaStats.get('groups')); // [{class: idx, sum: area}, ...]
  var fcGroups = ee.FeatureCollection(groups.map(function(g) {
    var d = ee.Dictionary(g);
    return ee.Feature(null, {class: d.get('class'), area: d.get('sum')});
  }));
  
  // Sort by class index and extract arrays
  var fcSorted = fcGroups.sort('class');
  var presentIdx = fcSorted.aggregate_array('class'); // ee.List
  var sumsByIdx = fcSorted.aggregate_array('area');   // ee.List
  
  // Build labels only for present classes: "idx: name"
  var wcNamesList = ee.List(wcNames); // server-ize names
  var labelsPresent = presentIdx.map(function(idx){
    idx = ee.Number(idx);
    var idxStr = idx.format();
    var nameStr = ee.String(wcNamesList.get(idx));
    return ee.String(idxStr).cat(': ').cat(nameStr);
  });
  
  return classified;
}

  .sqrt()
  .multiply(30)
  .rename('dist_coast');

  '(d < 500) ? 4 : (d < 1000) ? 3 : (d < 2000) ? 2 : 1',
  {d: distanceToCoast}
).rename('distance_risk');
// 1=Low (>=2000m), 2=Moderate (<2000m), 3=High (<1000), 4=Extreme (<500m)

var L_elevationRisk = lidar.select('dtm').expression(
  '(b < 2) ? 4 : (b < 5) ? 3 : (b < 10) ? 2 : 1',
  {b: lidarYS.select('dtm')}
).rename('elevation_risk_lidar');
// 1=Low (10m), 2=Moderate (5-10m), 3=High (2-5m), 4=Extreme (<2m)

// create coastal erosion risk weights by land cover
function landcover_to_risk(landcover_img){
  var L_landcoverRisk = landcover_img.expression(
    '(b == 4) ? 5' +        // Built-up
    ' : (b == 3) ? 4' +     // Cropland
    ' : (b == 2) ? 3' +     // Grassland
    ' : (b == 1) ? 2' +     // Shrubland
    ' : 1',
    {b: landcover_img}
  ).rename('landcover_risk_lidar')
  
  return L_landcoverRisk
}
// 1=Low (Everything else), 2=Moderate/Low (Shrubland), 3=Moderate (Grassland), 4=High (Cropland), 5=Extreme (Built-Up)

function get_vulnerability(landcover_risk){
var L_vulnerability = landcover_risk.multiply(0.50)
  .add(L_elevationRisk.multiply(0.30)
  .add(distanceRisk.multiply(0.20)))
  .rename('LiDAR vulnerability')
  .clip(yorkshire);
  
  return L_vulnerability;
}
